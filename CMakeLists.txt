cmake_minimum_required(VERSION 3.22)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Debug")
endif()

set(CMAKE_PROJECT_NAME ONMonlychassias)
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

project(${CMAKE_PROJECT_NAME})
message("Build type: " ${CMAKE_BUILD_TYPE})

enable_language(C ASM)

add_executable(${CMAKE_PROJECT_NAME})

add_subdirectory(cmake/stm32cubemx)

target_link_directories(${CMAKE_PROJECT_NAME} PRIVATE
)

target_sources(${CMAKE_PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/BSP/BSP.c
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/CAN1/CAN1.c
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/CAN2/CAN2.c
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/TIM2/TIM2.c
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/TIM6/TIM6.c
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/USART1/USART1.c
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/USART3/USART3.c
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/USART6/USART6.c
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Remote_Task/Remote_Task.c
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/CAN_Send_Task/CAN_Send_Task.c
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Control_Task/Control_Task.c
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Chassis_Task/Chassis_Task.c
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Gimbal_Task/Gimbal_Task.c
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Shooter_Task/Shooter_Task.c
    ${CMAKE_CURRENT_SOURCE_DIR}/Judge_System/Judge_System.c
    ${CMAKE_CURRENT_SOURCE_DIR}/Control_Algorithm/PID/PID.c
)

target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/BSP
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/CAN1
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/CAN2
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/TIM2
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/TIM6
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/USART1
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/USART3
    ${CMAKE_CURRENT_SOURCE_DIR}/SYSTEM/USART6
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Remote_Task
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/CAN_Send_Task
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Control_Task
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Chassis_Task
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Gimbal_Task
    ${CMAKE_CURRENT_SOURCE_DIR}/TASK/Shooter_Task
    ${CMAKE_CURRENT_SOURCE_DIR}/Judge_System
    ${CMAKE_CURRENT_SOURCE_DIR}/Control_Algorithm/PID
)

target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE
)

list(REMOVE_ITEM CMAKE_C_IMPLICIT_LINK_LIBRARIES ob)

target_link_libraries(${CMAKE_PROJECT_NAME}
    stm32cubemx
)
